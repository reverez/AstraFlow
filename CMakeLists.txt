cmake_minimum_required(VERSION 3.24)
project(AstraFlow VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(ASTRAFLOW_ENABLE_CUDA "Build CUDA production backend" ON)
option(ASTRAFLOW_BUILD_GUI "Build interactive interface" OFF)
option(ASTRAFLOW_BUILD_TESTS "Build tests" ON)
option(ASTRAFLOW_BUILD_BENCHMARKS "Build performance benchmarks" OFF)
include(FetchContent)
find_package(Threads REQUIRED)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(json)
add_library(astraflow src/analysis/convergence.cpp src/core/device.cpp src/cpu/euler1d.cpp src/core/solver.cpp src/core/simulation.cpp src/core/controller.cpp src/visualization/fields.cpp src/io/config.cpp src/io/output.cpp src/analysis/engineering.cpp src/geometry/mesh.cpp src/geometry/nozzle.cpp src/cpu/solver.cpp)
target_include_directories(astraflow PUBLIC include)
target_link_libraries(astraflow PUBLIC nlohmann_json::nlohmann_json Threads::Threads)
target_compile_options(astraflow PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic;-ffp-contract=off>)
if(ASTRAFLOW_ENABLE_CUDA)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 120)
  endif()
  enable_language(CUDA)
  set(CMAKE_CUDA_STANDARD 20)
  set(CMAKE_CUDA_STANDARD_REQUIRED ON)
  find_package(CUDAToolkit 12.8 REQUIRED)
  set_target_properties(astraflow PROPERTIES CUDA_ARCHITECTURES "${CMAKE_CUDA_ARCHITECTURES}" CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED ON)
  target_compile_options(astraflow PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
  target_sources(astraflow PRIVATE src/cuda/device.cu src/cuda/euler1d.cu src/cuda/solver.cu)
  target_compile_definitions(astraflow PUBLIC ASTRAFLOW_HAS_CUDA=1)
  target_link_libraries(astraflow PUBLIC CUDA::cudart)
endif()
add_executable(astraflow_cli apps/cli/main.cpp)
target_link_libraries(astraflow_cli PRIVATE astraflow)
if(ASTRAFLOW_BUILD_TESTS)
  include(CTest)
  FetchContent_Declare(Catch2 URL https://github.com/catchorg/Catch2/archive/refs/tags/v3.7.1.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(Catch2)
  add_executable(astraflow_tests tests/unit/convergence.cpp tests/unit/environment.cpp tests/unit/io.cpp tests/unit/controller.cpp tests/unit/euler.cpp tests/unit/mesh.cpp tests/verification/nozzle.cpp tests/verification/viscous.cpp tests/verification/refinement.cpp)
  target_link_libraries(astraflow_tests PRIVATE astraflow Catch2::Catch2WithMain)
  add_test(NAME unit COMMAND astraflow_tests "[unit]")
  add_test(NAME grid_refinement COMMAND astraflow_tests "[refinement]")
  add_test(NAME viscous_verification COMMAND astraflow_tests "[viscous]")
  add_test(NAME nozzle_verification COMMAND astraflow_tests "[nozzle]")
  add_test(NAME verification COMMAND astraflow_tests "[verification]")
  if(ASTRAFLOW_ENABLE_CUDA)
    target_sources(astraflow_tests PRIVATE tests/verification/cuda_euler.cpp)
    add_test(NAME cuda_verification COMMAND astraflow_tests "[cuda]")
    add_test(NAME cuda_device COMMAND astraflow_cli --device-info)
  endif()
  add_test(NAME cli_help COMMAND astraflow_cli --help)
endif()

if(ASTRAFLOW_BUILD_GUI)
  include(cmake/Gui.cmake)
endif()

if(ASTRAFLOW_BUILD_BENCHMARKS)
  add_executable(astraflow_benchmark benchmarks/solver.cpp)
  target_link_libraries(astraflow_benchmark PRIVATE astraflow)
endif()
